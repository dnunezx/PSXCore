cmake_minimum_required(VERSION 3.16)
project(PSXCore VERSION 0.1.0 LANGUAGES C)

add_library(psxcore STATIC src/vcd.c src/request.c src/elf.c src/boot_elf.c src/boot_sha256.c
    src/compat.c src/launch.c src/patch.c src/runtime_corrections.c src/discmap.c src/sha1.c)
add_library(PSXCore::core ALIAS psxcore)
target_include_directories(psxcore PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_compile_features(psxcore PUBLIC c_std_11)

if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
    include(CTest)
    if(BUILD_TESTING)
        add_executable(test_memory_access tests/test_memory_access.c runtime/memoryaccess.c)
        add_test(NAME memory_access_emitter COMMAND test_memory_access)
        add_executable(test_profiles tests/test_profiles.c)
        target_link_libraries(test_profiles PRIVATE PSXCore::core)
        add_test(NAME compatibility_profiles COMMAND test_profiles)
        add_executable(test_runtime_corrections tests/test_runtime_corrections.c)
        target_link_libraries(test_runtime_corrections PRIVATE PSXCore::core)
        add_test(NAME runtime_corrections COMMAND test_runtime_corrections)
    endif()
endif()
